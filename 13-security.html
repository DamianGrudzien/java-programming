Spring Security
& JWT

Autentykacja i autoryzacja w aplikacjach Spring Boot. Bezstanowe API z JWT, ochrona endpointów, hashowanie haseł i podstawy OAuth2.

Spring Security 6 Poziom: Zaawansowany
🔐
Autentykacja vs Autoryzacja: Autentykacja = Kim jesteś? (login/hasło, JWT). Autoryzacja = Co możesz robić? (role, uprawnienia). Spring Security obsługuje obie warstwy.

Konfiguracja Security — SecurityFilterChain

SecurityConfig.java — Spring Security 6 
@Configuration
@EnableWebSecurity
@EnableMethodSecurity  // włącza @PreAuthorize na metodach
public class SecurityConfig {

    private final JwtAuthFilter jwtFilter;

    @Bean
    public SecurityFilterChain filterChain(HttpSecurity http) throws Exception {
        return http
            // ✅ Bezstanowe API — bez sesji HTTP
            .sessionManagement(s -> s
                .sessionCreationPolicy(SessionCreationPolicy.STATELESS))

            // Wyłącz CSRF dla REST API (tokeny JWT chronią wystarczająco)
            .csrf(AbstractHttpConfigurer::disable)

            // Reguły autoryzacji — od najbardziej szczegółowych do ogólnych!
            .authorizeHttpRequests(auth -> auth
                .requestMatchers("/api/auth/**").permitAll()          // publiczne
                .requestMatchers("/api/admin/**").hasRole("ADMIN")   // tylko admin
                .requestMatchers(HttpMethod.GET, "/api/produkty")
                    .hasAnyRole("USER", "ADMIN")
                .anyRequest().authenticated()                        // reszta wymaga auth
            )

            // Dodaj filtr JWT przed filtrem username/password
            .addFilterBefore(jwtFilter, UsernamePasswordAuthenticationFilter.class)
            .build();
    }

    // Encoder haseł — BCrypt to standard (cost factor 12 zalecany)
    @Bean
    public PasswordEncoder passwordEncoder() {
        return new BCryptPasswordEncoder(12);
    }

    @Bean
    public AuthenticationManager authManager(AuthenticationConfiguration cfg)
            throws Exception {
        return cfg.getAuthenticationManager();
    }
}

JWT — JSON Web Token

JWT to podpisany token zawierający claims (twierdzenia) o użytkowniku. Składa się z trzech części: header.payload.signature zakodowanych w Base64. Serwer weryfikuje podpis — nie potrzebuje sesji ani bazy.

JwtSerwis.java — generowanie i weryfikacja tokenów 
@Service
public class JwtSerwis {

    @Value("${app.jwt.secret}")
    private String secret;       // min 256-bit — ze zmiennej środowiskowej!

    @Value("${app.jwt.expiration}")
    private long expirationSec;   // np. 3600 = 1 godzina

    public String generuj(UserDetails user) {
        var teraz = Instant.now();
        return Jwts.builder()
            .subject(user.getUsername())
            .claim("roles", user.getAuthorities()
                .stream().map(GrantedAuthority::getAuthority).toList())
            .issuedAt(Date.from(teraz))
            .expiration(Date.from(teraz.plusSeconds(expirationSec)))
            .signWith(klucz(), Jwts.SIG.HS256)
            .compact();
    }

    public Claims parsuj(String token) {
        return Jwts.parser()
            .verifyWith(klucz())
            .build()
            .parseSignedClaims(token)
            .getPayload();
    }

    public boolean czyWazny(String token, UserDetails user) {
        try {
            var claims = parsuj(token);
            return claims.getSubject().equals(user.getUsername())
                && !claims.getExpiration().before(new Date());
        } catch (JwtException e) {
            return false;
        }
    }

    private SecretKey klucz() {
        return Keys.hmacShaKeyFor(Decoders.BASE64.decode(secret));
    }
}

Filtr JWT — Każde Żądanie

JwtAuthFilter.java 
@Component
@RequiredArgsConstructor
public class JwtAuthFilter extends OncePerRequestFilter {

    private final JwtSerwis jwt;
    private final UserDetailsService uds;

    @Override
    protected void doFilterInternal(
            HttpServletRequest req,
            HttpServletResponse res,
            FilterChain chain) throws ServletException, IOException {

        var header = req.getHeader("Authorization");

        // Brak tokenu lub zły format — pomiń filtr
        if (header == null || !header.startsWith("Bearer ")) {
            chain.doFilter(req, res);
            return;
        }

        var token = header.substring(7); // usuń "Bearer "
        var username = jwt.parsuj(token).getSubject();

        if (username != null && SecurityContextHolder.getContext().getAuthentication() == null) {
            var user = uds.loadUserByUsername(username);

            if (jwt.czyWazny(token, user)) {
                var auth = new UsernamePasswordAuthenticationToken(
                    user, null, user.getAuthorities());
                auth.setDetails(new WebAuthenticationDetailsSource().buildDetails(req));
                // Ustaw użytkownika w SecurityContext — żądanie jest teraz uwierzytelnione
                SecurityContextHolder.getContext().setAuthentication(auth);
            }
        }
        chain.doFilter(req, res);
    }
}

Autoryzacja na Poziomie Metody

@PreAuthorize — granularna kontrola dostępu 
@RestController
@RequestMapping("/api/admin")
public class AdminController {

    // Tylko użytkownicy z rolą ADMIN
    @GetMapping("/uzytkownicy")
    @PreAuthorize("hasRole('ADMIN')")
    public List<UzytkownikDto> wszyscyUzytkownicy() { ... }

    // Użytkownik może pobrać SWÓJ profil LUB admin każdy
    @GetMapping("/uzytkownicy/{id}")
    @PreAuthorize("hasRole('ADMIN') or #id == authentication.principal.id")
    public UzytkownikDto profil(@PathVariable Long id) { ... }

    // SpEL — Spring Expression Language w @PreAuthorize
    @DeleteMapping("/uzytkownicy/{id}")
    @PreAuthorize("hasRole('SUPER_ADMIN') and #id != authentication.principal.id")
    public void usunUzytkownika(@PathVariable Long id) { ... }
}

// Pobranie aktualnie zalogowanego użytkownika
@GetMapping("/moj-profil")
public UzytkownikDto mojProfil(@AuthenticationPrincipal UserDetails user) {
    return serwis.znajdz(user.getUsername());
}

Endpoint Logowania — AuthController

AuthController.java 
@RestController
@RequestMapping("/api/auth")
@RequiredArgsConstructor
public class AuthController {

    private final AuthenticationManager authManager;
    private final JwtSerwis jwtSerwis;
    private final UserDetailsService uds;

    // POST /api/auth/login
    @PostMapping("/login")
    public ResponseEntity<TokenResponse> login(@RequestBody LoginRequest req) {
        // Rzuci BadCredentialsException jeśli dane nieprawidłowe
        authManager.authenticate(
            new UsernamePasswordAuthenticationToken(req.login(), req.haslo()));

        var user = uds.loadUserByUsername(req.login());
        var token = jwtSerwis.generuj(user);

        return ResponseEntity.ok(new TokenResponse(token, "Bearer", 3600));
    }
}

// Request/Response jako Records
public record LoginRequest(@NotBlank String login, @NotBlank String haslo) {}
public record TokenResponse(String token, String typ, long expiresIn) {}
⚠️
Bezpieczeństwo haseł: Nigdy nie przechowuj haseł jako plaintext. Używaj BCrypt z cost factor ≥ 10. Nigdy nie loguj haseł. Używaj HTTPS w produkcji. Przechowuj JWT secret w zmiennych środowiskowych, nie w kodzie.
